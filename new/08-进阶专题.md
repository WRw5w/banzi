# 08. 银牌常见进阶与生僻板子

这些算法不一定单独构成 2200 以下题，但经常作为区域赛银牌题的关键子步骤。需要时直接查本页和 `templates/competitive.hpp`。

## 1. 线性基（XOR Basis）

维护最高位不同的向量。插入时从高位向下消元；最大异或值从高位贪心尝试。可扩展为求第 k 小异或、判断异或方程组是否有解、求线性空间维数。

```cpp
struct LinearBasis {
    long long b[61]{}; // b[i] 的最高位是 i，每个最高位只保留一个向量。
    bool insert(long long x) {
        for (int i = 60; i >= 0; --i) if (x >> i & 1) {
            // 从高位消元，保证插入后的基按最高位三角化。
            if (!b[i]) { b[i] = x; return true; }
            x ^= b[i];
        }
        // x 被完全消成 0，说明它是已有向量的线性组合。
        return false;
    }
    long long maximum(long long x = 0) const {
        // 从高位向低位贪心：异或后变大就保留，低位不能破坏高位最优。
        for (int i = 60; i >= 0; --i) x = max(x, x ^ b[i]);
        return x;
    }
};
```

## 2. 三维偏序的 CDQ 分治（CDQ Divide and Conquer）

把三维偏序/带时间的 DP 按一维分治，左侧贡献右侧，第三维用树状数组维护，处理完后撤销修改。典型复杂度 `O(n log²n)`；重点是“只让左半影响右半”和清空 BIT。

```cpp
struct Point3D {
    int x, y, z, id;
    long long ans = 0;
};
vector<Point3D> pts, buf;
struct BIT {
    vector<int> t;
    BIT(int n): t(n + 1) {}
    void add(int x, int v) {
        // z 必须先离散化为 1..MAXZ；0 不能作为树状数组下标。
        for (; x < (int)t.size(); x += x & -x) t[x] += v;
    }
    int sum(int x) const {
        // 返回 z<=x 的计数。
        int r = 0; for (; x; x -= x & -x) r += t[x]; return r;
    }
} bit(MAXZ);

// 统计每个点左下方（x,y,z 均 <=）的点数；初始 pts 必须按 x,y,z 排序
void cdq(int l, int r) {
    // 分治下标代表第一维 x：左半的点只能给右半贡献。
    if (l == r) return;
    int m = (l + r) / 2;
    cdq(l, m); cdq(m + 1, r); // 返回时左右段分别按 y 有序。
    int i = l;
    for (int j = m + 1; j <= r; ++j) {
        // 双指针处理第二维 y；已经满足的左半点加入 BIT。
        while (i <= m && pts[i].y <= pts[j].y) bit.add(pts[i++].z, 1);
        // BIT 前缀和处理第三维 z。
        pts[j].ans += bit.sum(pts[j].z);
    }
    // 撤销 BIT，防止其它分治区间读到本次贡献。
    for (int k = l; k < i; ++k) bit.add(pts[k].z, -1);
    // 合并成按 y 有序，供父层继续双指针。
    inplace_merge(pts.begin() + l, pts.begin() + m + 1,
                  pts.begin() + r + 1,
                  [](const Point3D& a, const Point3D& b) { return a.y < b.y; });
}

sort(pts.begin(), pts.end(), [](const Point3D& a, const Point3D& b) {
    return tie(a.x, a.y, a.z) < tie(b.x, b.y, b.z);
});
cdq(0, (int)pts.size() - 1);
```

## 3. 整体二分

多个询问同时二分答案；对当前值域中点，把所有操作/询问按时间递归，使用树状数组或线段树统计 `check`。适合“第 k 次出现/第 k 小”且操作可离线，复杂度常为 `O((n+q)log V log n)`。

```cpp
void solve(vector<int> ids, int lo, int hi) {
    // ids 是当前仍可能落在答案值域 [lo,hi] 的询问集合。
    if (ids.empty()) return;
    if (lo == hi) {
        for (int id : ids) answer[id] = lo;
        return;
    }
    int mid = (lo + hi) / 2; vector<int> left, right;
    // 把“答案<=mid”对应的操作加入数据结构，然后统一检查所有询问。
    apply_operations(mid);
    for (int id : ids) (check(query[id]) ? left : right).push_back(id);
    // 回滚后，左右递归必须从同一个历史状态开始。
    rollback_to_start_of_round();
    solve(left, lo, mid); solve(right, mid + 1, hi);
}
```

## 4. 线段树分治时间

把每条边/每个事件存在的时间区间加入线段树节点，DFS 线段树时用回滚 DSU 维护当前时间点状态，离开节点回滚。解决动态连通性、区间激活约束。

```cpp
void solve_time(int p, int l, int r) {
    // 当前递归节点代表时间区间 [l,r]；seg[p] 中的事件在整个区间都有效。
    int snap = dsu.snapshot();
    for (auto [u, v] : seg[p]) dsu.unite(u, v);
    if (l == r) {
        // 到叶子时，DSU 恰好包含时间 l 的所有活动边。
        answer[l] = dsu.connected(query[l].u, query[l].v);
    }
    else {
        int m = (l + r) / 2;
        solve_time(p * 2, l, m); solve_time(p * 2 + 1, m + 1, r);
    }
    // 离开节点必须恢复进入前的版本，保证兄弟时间段互不污染。
    dsu.rollback(snap);
}
```

## 5. 点分治

每层找重心，把跨重心的路径贡献按子树统计并容斥，递归子树。统计距离、颜色、路径权值时要设计可合并的信息；重心分解树还能支持在线修改/查询。

```cpp
int get_centroid(int u, int p, int total) {
    // 如果某个未删除儿子超过当前连通块一半，重心必在该儿子方向。
    for (int v : tree[u]) if (v != p && !removed[v])
        if (sz[v] * 2 > total) return get_centroid(v, u, total);
    return u;
}
void decompose(int entry) {
    // 当前层先找重心、记录跨重心贡献，再递归处理删除重心后的子块。
    int c = get_centroid(entry, 0, calc_size(entry, 0));
    removed[c] = true;
    for (int v : tree[c]) if (!removed[v]) decompose(v);
}
```

## 6. 虚树

给定树上少量关键点，只保留关键点及其两两相邻 LCA。按 DFS 序排序并用栈建边，节点数 `O(k)`；在虚树上做 DP，常用于多次询问的“关键点之间代价”。

```cpp
sort(key.begin(), key.end(), [&](int a, int b) { return tin[a] < tin[b]; });
vector<int> nodes = key;
// 相邻关键点的 LCA 足以补全虚树的全部分叉点。
for (int i = 0, m = key.size(); i + 1 < m; ++i) nodes.push_back(lca(key[i], key[i + 1]));
sort(nodes.begin(), nodes.end(), [&](int a, int b) { return tin[a] < tin[b]; });
nodes.erase(unique(nodes.begin(), nodes.end()), nodes.end());
vector<pair<int, int>> virtual_edges;
vector<int> st;
for (int u : nodes) {
    // 栈内保持当前 DFS 链；不是祖先就弹出。
    while (!st.empty() && !is_ancestor(st.back(), u)) st.pop_back();
    if (!st.empty()) virtual_edges.push_back({st.back(), u});
    st.push_back(u);
}
```

## 7. 斯坦纳树（了解）

终端点数量 `k` 较小时，`dp[mask][v]` 表示连接 mask 终端且以 v 为根的最小代价；子集合并后跑最短路。复杂度约 `O(3^k V + 2^k E log V)`。

```cpp
for (int mask = 1; mask < (1 << k); ++mask) {
    // mask 拆成两个非空子集，在同一个 v 汇合。
    for (int sub = (mask - 1) & mask; sub; sub = (sub - 1) & mask)
        for (int v = 0; v < n; ++v)
            dp[mask][v] = min(dp[mask][v], dp[sub][v] + dp[mask ^ sub][v]);
    // 子集合并后，末端 v 还可以沿图边移动；固定 mask 跑一次最短路。
    dijkstra_on_state(mask); // 用图的边松弛 dp[mask][*]
}
```

## 8. 二可满足性算法（2-Satisfiability，2-SAT）

每个布尔变量两个节点，子句 `(a||b)` 建 `¬a→b`、`¬b→a`；SCC 后若变量与其否定同 SCC 则无解，否则按 SCC 拓扑序赋值。适合“二选一且两者不能同时为假”的约束。

```cpp
auto add_clause = [&](int a, bool va, int b, bool vb) {
    // literal 编号：2*i+0 / 2*i+1 是同一变量的两个取值，^1 表示取反。
    int x = 2 * a + va, y = 2 * b + vb;
    // (x OR y) 等价于 (!x -> y) AND (!y -> x)。
    g[x ^ 1].push_back(y); g[y ^ 1].push_back(x);
};
// (x_a == va) OR (x_b == vb)
for (int i = 0; i < vars; ++i) {
    // 同一变量和其否定在同一 SCC 时，互相推出矛盾，无解。
    if (scc[2 * i] == scc[2 * i + 1]) impossible = true;
    // SCC 拓扑顺序方向取决于实现，使用 Tarjan/Kosaraju 时要核对这一行。
    else assignment[i] = scc[2 * i] > scc[2 * i + 1];
}
```

## 9. 区间最小值/最大值的笛卡尔树

以数组最小值为根，左/右递归构成笛卡尔树；单调栈 `O(n)` 构造。可把 RMQ/LCA、子数组最小值贡献、Treap 结构统一起来。

```cpp
vector<int> left(n, -1), right(n, -1), st;
for (int i = 0; i < n; ++i) {
    int last = -1;
    // 单调栈维护当前最小笛卡尔树的右链。
    while (!st.empty() && a[st.back()] > a[i]) last = st.back(), st.pop_back();
    // 右链最后一个没有被弹出的节点把 i 作为右儿子。
    if (!st.empty()) right[st.back()] = i;
    // 被弹出的整段节点成为 i 的左子树。
    left[i] = last; st.push_back(i);
}
// 栈底是根；若想避免重复找根，构造时也可以记录第一个未被挂为右儿子的节点。
int root = st.front();
while (left[root] != -1) root = left[root];
```

## 10. 二进制提升的更多用法

不只求 LCA：沿跳跃表同时维护最大边、最小边、边权和、函数迭代。函数图（每点出度 1）先找环，再用倍增回答第 k 步位置，复杂度 `O((n+q)log K)`。

```cpp
int jump(int u, long long k) {
    // up[u][j] 表示从 u 跳 2^j 步后的位置；LOG 要覆盖 k 的最高位。
    for (int j = 0; j < LOG; ++j) if (k >> j & 1) u = up[u][j];
    return u;
}
```

## 11. 置换与循环

置换分解成不相交环；重复应用 `k` 次时每个位置在所属环内移动 `k mod len`。求置换逆、循环节、最少交换次数都从环分解出发。

```cpp
vector<int> vis(n), result(n);
for (int i = 0; i < n; ++i) if (!vis[i]) {
    vector<int> cyc;
    for (int u = i; !vis[u]; u = p[u]) vis[u] = 1, cyc.push_back(u);
    int m = cyc.size();
    for (int j = 0; j < m; ++j) result[cyc[j]] = cyc[(j + k) % m];
}
```

## 12. 快速傅里叶变换与数论变换（了解）

多项式卷积、计数合并、字符串匹配。NTT 要求模数为 `c·2^k+1`（常用 `998244353`），每层蝶形乘原根；逆变换乘 `n^{-1}`。只在卷积规模大于 `O(n²)` 时值得使用。

```cpp
void ntt(vector<int>& a, bool inv) {
    int n = a.size();
    // 先做位逆序排列，把同一蝶形层需要的数据放到相邻位置。
    for (int i = 1, j = 0; i < n; ++i) {
        int bit = n >> 1; for (; j & bit; bit >>= 1) j ^= bit; j ^= bit;
        if (i < j) swap(a[i], a[j]);
    }
    for (int len = 2; len <= n; len <<= 1) {
        // len 是当前蝶形块长度；wlen 是该层的单位根。
        int wlen = qpow(G, (MOD - 1) / len);
        if (inv) wlen = qpow(wlen, MOD - 2);
        for (int i = 0; i < n; i += len) {
            long long w = 1;
            for (int j = 0; j < len / 2; ++j) {
                int u = a[i + j], v = w * a[i + j + len / 2] % MOD;
                // 蝶形：上路 u+v，下路 u-v；w 每次乘单位根。
                a[i + j] = (u + v) % MOD; a[i + j + len / 2] = (u - v + MOD) % MOD;
                w = w * wlen % MOD;
            }
        }
    }
    if (inv) {
        // 正变换的逆需要再乘 n^{-1}；逆单位根已经在上面处理。
        long long ni = qpow(n, MOD - 2);
        for (int &x : a) x = x * ni % MOD;
    }
}
```

## 13. 高斯消元

解线性方程组、求秩、行列式。整数模数下用模逆；浮点版本选主元避免除以接近 0 的数。方程有自由变量时记录主元列并构造一组解。

```cpp
int gauss(vector<vector<long long>> a) {
    // 返回秩；a 若为增广矩阵，矛盾行还要额外判断。
    int n = a.size(), m = a[0].size(), rank = 0;
    for (int col = 0; col < m && rank < n; ++col) {
        int pivot = rank;
        // 找当前列第一个非零主元；实数版应换成绝对值最大。
        while (pivot < n && a[pivot][col] == 0) ++pivot;
        if (pivot == n) continue;
        swap(a[pivot], a[rank]);
        // 模质数下除主元等价于乘费马逆元。
        long long inv = qpow(a[rank][col], MOD - 2);
        for (int j = col; j < m; ++j) a[rank][j] = a[rank][j] * inv % MOD;
        for (int i = 0; i < n; ++i) if (i != rank && a[i][col]) {
            long long t = a[i][col];
            // Gauss-Jordan 把其它行的当前列清零。
            for (int j = col; j < m; ++j) a[i][j] = (a[i][j] - t * a[rank][j]) % MOD;
        }
        ++rank;
    }
    return rank;
}
```

## 14. 匈牙利算法与 Kuhn–Munkres 加权匹配算法（Hungarian / KM）

KM 求完备二分图最大权匹配 `O(n³)`，节点数通常几百以内。若只是无权最大匹配用 Hopcroft–Karp；若是最小费用流模型则用费用流更灵活。

```cpp
// Hungarian：最小费用完备匹配，a[i][j] 为费用，行列从 1 开始
vector<long long> u(n + 1), v(n + 1); vector<int> p(n + 1), way(n + 1);
for (int i = 1; i <= n; ++i) {
    // 当前把第 i 行加入；p[j] 表示当前匹配中列 j 匹配的行。
    p[0] = i; int j0 = 0; vector<long long> mn(n + 1, INF); vector<char> used(n + 1);
    do {
        // 从当前交替树的列 j0 扩展未使用列。
        used[j0] = true; int i0 = p[j0], j1 = 0; long long delta = INF;
        for (int j = 1; j <= n; ++j) if (!used[j]) {
            // reduced cost；势能 u/v 保证它非负。
            long long cur = a[i0][j] - u[i0] - v[j];
            if (cur < mn[j]) mn[j] = cur, way[j] = j0;
            if (mn[j] < delta) delta = mn[j], j1 = j;
        }
        // 调整势能，让至少一条新的边 reduced cost 变成 0。
        for (int j = 0; j <= n; ++j) if (used[j]) u[p[j]] += delta, v[j] -= delta; else mn[j] -= delta;
        j0 = j1;
    } while (p[j0]);
    // 沿 way 回溯增广路，反转匹配关系。
    do { int j1 = way[j0]; p[j0] = p[j1]; j0 = j1; } while (j0);
}
```

## 15. 约束满足与状态图

- “位置 + 剩余次数/钥匙/开关”→ 分层图或状态 BFS。
- “先后依赖 + 最少时间”→ 拓扑 DP。
- “有向图中每步应用函数”→ 倍增。
- “所有子集的某种聚合”→ SOS DP/子集卷积（先确认是否真的需要）。

```cpp
struct State { int u, keys, dist; };
queue<State> q;
set<pair<int, int>> seen;
q.push({start, 0, 0}); seen.insert({start, 0});
while (!q.empty()) {
    auto cur = q.front(); q.pop();
    for (auto [v, need, gain] : transitions[cur.u]) {
        if ((cur.keys & need) != need) continue;
        int nk = cur.keys | gain;
        if (seen.insert({v, nk}).second) q.push({v, nk, cur.dist + 1});
    }
}
```
